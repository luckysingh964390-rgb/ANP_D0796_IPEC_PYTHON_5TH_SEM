students = []

while True:
    print("1.Add 2.Display 3.Search 4.Exit")
    ch = int(input("Choice: "))

    if ch == 1:
        r = input("Roll: ")
        n = input("Name: ")
        a = input("Age: ")
        c = input("Course: ")
        m = int(input("Marks: "))
        students.append({"Roll": r, "Name": n, "Age": a, "Course": c, "Marks": m})

    elif ch == 2:
        for i in students:
            print(i)

    elif ch == 3:
        r = input("Enter Roll: ")
        for i in students:
            if i["Roll"] == r:
                print(i)

    elif ch == 4:
        break

    else:
        print("Invalid Choice")
